<script>
    function openCreateModal() {
        document.getElementById('modalTitle').textContent = 'Nouvelle province';
        document.getElementById('province_id').value = '';
        document.getElementById('province_name').value = '';
        document.getElementById('chef_lieu').value = '';
        document.getElementById('provinceForm').action = "{% url 'province-create' %}";
        document.getElementById('provinceModal').classList.remove('hidden');
    }
    
    function editProvince(id, name) {
        document.getElementById('modalTitle').textContent = 'Modifier la province';
        document.getElementById('province_id').value = id;
        document.getElementById('province_name').value = name;
        document.getElementById('provinceForm').action = `/provinces/${id}/edit/`;
        document.getElementById('provinceModal').classList.remove('hidden');
    }
    
    function deleteProvince(id) {
        if (confirm('Êtes-vous sûr de vouloir supprimer cette province ?')) {
            window.location.href = `/provinces/${id}/delete/`;
        }
    }
    
    function closeModal() {
        document.getElementById('provinceModal').classList.add('hidden');
    }
    
    // Fermer le modal en cliquant en dehors
    window.onclick = function(event) {
        const modal = document.getElementById('provinceModal');
        if (event.target == modal) {
            closeModal();
        }
    }
</script>
